练习 2.2：创建 TypeScript 项目并实现圆面积计算

步骤：

1. 创建项目目录并初始化：

mkdir circle-project
cd circle-project
npm init -y
npm install typescript --save-dev
npx tsc --init

2. 创建目录结构：

circle-project/ ├── src/ │ └── index.ts ├── dist/ ├── tsconfig.json └── package.json

3. src/index.ts 代码：

// 计算圆面积的函数
function calculateCircleArea(radius: number): number {
  // 圆面积公式：π × r²
  const pi = Math.PI;
  return pi * radius * radius;
}

// 测试
const radius1 = 5;
const area1 = calculateCircleArea(radius1);
console.log(`半径为 ${radius1} 的圆面积: ${area1.toFixed(4)}`);

const radius2 = 10;
const area2 = calculateCircleArea(radius2);
console.log(`半径为 ${radius2} 的圆面积: ${area2.toFixed(4)}`);

// 计算多个圆的面积
const radii = [1, 2, 3, 4, 5];
console.log("\n多个圆的面积：");
radii.forEach(r => {
  const area = calculateCircleArea(r);
  console.log(`r = ${r}: A = ${area.toFixed(4)}`);
});

4. 编译并运行：

npx tsc
node dist/index.js

运行结果：

半径为 5 的圆面积: 78.5398
半径为 10 的圆面积: 314.1593

多个圆的面积：
r = 1: A = 3.1416
r = 2: A = 12.5664
r = 3: A = 28.2743
r = 4: A = 50.2655
r = 5: A = 78.5398

练习 2.3：配置 tsconfig.json

要求：

  • 编译目标为 ES2020
  • 启用严格模式
  • 输出目录为 `./dist`
  • 源文件目录为 `./src`
  • 生成声明文件（`.d.ts`）

配置文件：

{
  "compilerOptions": {
    /* 基本选项 */
    "target": "ES2020",
    "module": "commonjs",
    "lib": ["ES2020"],

    /* 输出配置 */
    "outDir": "./dist",
    "rootDir": "./src",

    /* 严格模式 */
    "strict": true,

    /* 生成声明文件 */
    "declaration": true,

    /* 模块解析 */
    "moduleResolution": "node",
    "esModuleInterop": true,

    /* 其他选项 */
    "sourceMap": true,
    "skipLibCheck": true,
    "forceConsistentCasingInFileNames": true
  },
  "include": ["src/**/*"],
  "exclude": ["node_modules", "dist"]
}
说明：
- "strict": true 启用所有严格类型检查
- "declaration": true 生成 .d.ts 声明文件
- "sourceMap": true 生成 .map 源映射文件，便于调试

练习 2.4：配置 npm 脚本

package.json scripts 配置：

{
  "name": "my-typescript-project",
  "version": "1.0.0",
  "main": "dist/index.js",
  "scripts": {
    "build": "tsc",
    "build:watch": "tsc -w",
    "start": "node dist/index.js",
    "clean": "rm -rf dist",
    "rebuild": "npm run clean && npm run build"
  },
  "devDependencies": {
    "typescript": "^5.3.3"
  }
}

使用方式：

# 编译项目
npm run build

# 监视模式编译（开发时使用）
npm run build:watch

# 运行编译后的文件
npm start

# 清理并重新编译
npm run rebuild
提示：start 脚本可以直接用 npm start 运行，不需要 run 命令。

项目结构：

my-project/ ├── src/ │ ├── utils/ │ │ ├── math.ts │ │ └── string.ts │ └── index.ts ├── dist/ │ ├── utils/ │ │ ├── math.js │ │ ├── math.d.ts │ │ ├── string.js │ │ └── string.d.ts │ ├── index.js │ ├── index.d.ts │ └── index.map ├── tsconfig.json └── package.json

src/utils/math.ts：

// 数学工具函数

export function add(a: number, b: number): number {
  return a + b;
}

export function subtract(a: number, b: number): number {
  return a - b;
}

export function multiply(a: number, b: number): number {
  return a * b;
}

export function divide(a: number, b: number): number {
  if (b === 0) {
    throw new Error("除数不能为零");
  }
  return a / b;
}

export function factorial(n: number): number {
  if (n < 0) {
    throw new Error("负数没有阶乘");
  }
  if (n <= 1) return 1;
  return n * factorial(n - 1);
}

src/utils/string.ts：

// 字符串工具函数

export function capitalize(str: string): string {
  if (!str) return "";
  return str.charAt(0).toUpperCase() + str.slice(1);
}

export function reverseString(str: string): string {
  return str.split("").reverse().join("");
}

export function truncate(str: string, maxLength: number): string {
  if (str.length <= maxLength) return str;
  return str.slice(0, maxLength) + "...";
}

export function countWords(str: string): number {
  return str.trim().split(/\s+/).filter(word => word.length > 0).length;
}

export function isPalindrome(str: string): boolean {
  const normalized = str.toLowerCase().replace(/[^a-z0-9]/g, "");
  return normalized === normalized.split("").reverse().join("");
}
