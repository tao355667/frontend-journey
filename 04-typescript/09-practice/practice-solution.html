练习 9.1：任务归档功能

题目：扩展任务服务，添加任务归档功能。

实现思路：

在 Task 接口中添加 `archived` 字段，添加 `archive()` 和 `unarchive()` 方法，并在过滤时默认排除已归档任务。

核心代码：

// 在 Task 接口中添加
export interface Task {
  // ... 其他字段
  archived?: boolean;
}

// 添加归档方法
archive(id: string): Task | undefined {
  return this.update(id, { archived: true });
}

unarchive(id: string): Task | undefined {
  return this.update(id, { archived: false });
}

// 过滤时默认排除已归档任务
filter(filter: TaskFilter): Task[] {
  let tasks = this.storage.load();
  
  // 默认排除已归档任务
  if (filter.archived === undefined) {
    filter.archived = false;
  }
  
  // ... 其他过滤逻辑
}

// 使用示例
app.archiveTask(task1.id);  // 归档任务
app.getTasks();             // 默认不包含已归档任务
app.getTasks({ archived: true });  // 显式查询已归档任务

// Comment 接口定义
export interface Comment {
  id: string;
  taskId: string;
  userId: string;
  content: string;
  createdAt: Date;
}

// Comment 服务
export class CommentService {
  private storage: StorageService;

  constructor() {
    this.storage = new StorageService("comments");
  }

  create(taskId: string, userId: string, content: string): Comment {
    const comment: Comment = {
      id: `comment-${Date.now()}`,
      taskId,
      userId,
      content,
      createdAt: new Date()
    };

    this.storage.addItem(comment);
    return comment;
  }

  getByTaskId(taskId: string): Comment[] {
    return this.storage.load().filter(c => c.taskId === taskId);
  }

  update(id: string, content: string): Comment | undefined {
    const comment = this.storage.getItem(id);
    if (!comment) return undefined;

    const updated = { ...comment, content };
    this.storage.updateItem(id, updated);
    return updated;
  }

  delete(id: string): boolean {
    return this.storage.deleteItem(id);
  }
}

// 在 TaskManagementApp 中使用
class TaskManagementApp {
  private commentService: CommentService;

  constructor() {
    // ...
    this.commentService = new CommentService();
  }

  addComment(taskId: string, userId: string, content: string): Comment {
    return this.commentService.create(taskId, userId, content);
  }

  getComments(taskId: string): Comment[] {
    return this.commentService.getByTaskId(taskId);
  }
}

// 使用示例
const comment = app.addComment(task1.id, user.id, "这是一个评论");

// 在 TaskService 中添加搜索方法
search(query: string): Task[] {
  if (!query || query.trim() === "") {
    return this.getAll();
  }

  const searchLower = query.toLowerCase().trim();

  return this.storage.load().filter(task => {
    // 搜索标题
    if (task.title.toLowerCase().includes(searchLower)) {
      return true;
    }

    // 搜索描述
    if (task.description.toLowerCase().includes(searchLower)) {
      return true;
    }

    // 搜索标签
    if (task.tags.some(tag => tag.toLowerCase().includes(searchLower))) {
      return true;
    }

    // 搜索指派人
    if (task.assignee && task.assignee.toLowerCase().includes(searchLower)) {
      return true;
    }

    return false;
  });
}

// 支持多关键词搜索
searchAdvanced(queries: string[]): Task[] {
  if (queries.length === 0) {
    return this.getAll();
  }

  return this.storage.load().filter(task => {
    return queries.every(query => {
      const searchLower = query.toLowerCase();
      return (
        task.title.toLowerCase().includes(searchLower) ||
        task.description.toLowerCase().includes(searchLower) ||
        task.tags.some(tag => tag.toLowerCase().includes(searchLower))
      );
    });
  });
}

// 使用示例
const results1 = app.searchTasks("TypeScript");
const results2 = app.searchTasks("学习 教程");
const results3 = app.searchAdvanced(["学习", "重要"]);

练习 9.4：任务排序功能

题目：实现任务排序功能。

实现代码：

// 排序类型
type SortBy = "created" | "dueDate" | "priority" | "status" | "title";
type SortOrder = "asc" | "desc";

// 优先级映射
const priorityOrder: Record = {
  urgent: 4,
  high: 3,
  medium: 2,
  low: 1
};

// 状态映射
const statusOrder: Record = {
  todo: 0,
  in_progress: 1,
  review: 2,
  done: 3
};

// 排序方法
sort(tasks: Task[], by: SortBy, order: SortOrder = "asc"): Task[] {
  const sorted = [...tasks].sort((a, b) => {
    let comparison = 0;

    switch (by) {
      case "created":
        comparison = a.createdAt.getTime() - b.createdAt.getTime();
        break;

      case "dueDate":
        // 处理无截止日期的情况
        if (!a.dueDate && !b.dueDate) comparison = 0;
        else if (!a.dueDate) comparison = 1;  // 无截止日期排后面
        else if (!b.dueDate) comparison = -1;
        else comparison = a.dueDate.getTime() - b.dueDate.getTime();
        break;

      case "priority":
        comparison = priorityOrder[a.priority] - priorityOrder[b.priority];
        break;

      case "status":
        comparison = statusOrder[a.status] - statusOrder[b.status];
        break;

      case "title":
        comparison = a.title.localeCompare(b.title, "zh-CN");
        break;
    }

    return order === "asc" ? comparison : -comparison;
  });

  return sorted;
}

// 组合过滤和排序
filterAndSort(filter: TaskFilter, sortBy: SortBy, sortOrder: SortOrder): Task[] {
  const filtered = this.filter(filter);
  return this.sort(filtered, sortBy, sortOrder);
}

// 使用示例
const tasks1 = app.sortTasks(allTasks, "priority", "desc");  // 按优先级降序
const tasks2 = app.sortTasks(allTasks, "created", "asc");    // 按创建时间升序
const tasks3 = app.sortTasks(allTasks, "dueDate", "asc");    // 按截止日期升序

JavaScript 实现：

// todo.js - 完整实现

class TodoApp {
  constructor() {
    this.todos = JSON.parse(localStorage.getItem("todos") || "[]");
    this.filter = "all";
    this.render();
  }

  add(title) {
    if (!title.trim()) return;
    
    const todo = {
      id: Date.now().toString(),
      title: title.trim(),
      completed: false,
      createdAt: new Date()
    };
    
    this.todos.unshift(todo);
    this.save();
    this.render();
  }

  toggle(id) {
    const todo = this.todos.find(t => t.id === id);
    if (todo) {
      todo.completed = !todo.completed;
      this.save();
      this.render();
    }
  }

  delete(id) {
    this.todos = this.todos.filter(t => t.id !== id);
    this.save();
    this.render();
  }

  save() {
    localStorage.setItem("todos", JSON.stringify(this.todos));
  }

  get filteredTodos() {
    switch (this.filter) {
      case "active":
        return this.todos.filter(t => !t.completed);
      case "completed":
        return this.todos.filter(t => t.completed);
      default:
        return this.todos;
    }
  }

  render() {
    const list = document.getElementById("todoList");
    list.innerHTML = this.filteredTodos.map(todo => `
      <div class="todo-item ${todo.completed ? 'completed' : ''}">
        <input type="checkbox" ${todo.completed ? "checked" : ""} 
          onchange="app.toggle('${todo.id}')">
        <span>${todo.title}</span>
        <button onclick="app.delete('${todo.id}')">删除</button>
      </div>
    `).join("");

    // 更新统计
    document.getElementById("totalCount").textContent = this.todos.length;
    document.getElementById("completedCount").textContent = 
      this.todos.filter(t => t.completed).length;

    // 更新过滤器按钮状态
    document.querySelectorAll(".filters button").forEach(btn => {
      btn.classList.remove("active");
    });
    document.getElementById(`btn-${this.filter}`).classList.add("active");
  }

  filterTodos(filter) {
    this.filter = filter;
    this.render();
  }
}

const app = new TodoApp();

function addTodo() {
  const input = document.getElementById("todoInput");
  app.add(input.value);
  input.value = "";
}

权限检查装饰器：

// 权限检查装饰器
function requirePermission(permission: keyof Permission) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;

    descriptor.value = function (...args: any[]) {
      const user = this.getCurrentUser();
      if (!user) {
        throw new Error("请先登录");
      }

      const permissions = RolePermissions[user.role];
      if (!permissions[permission]) {
        throw new Error(`权限不足：需要 ${permission} 权限`);
      }

      return originalMethod.apply(this, args);
    };

    return descriptor;
  };
}
