练习 7.2：交换元素

题目：编写一个泛型函数 `swap`，接受两个参数，返回交换后的元组。

TypeScript 代码：

// 交换两个元素
function swap(a: T, b: U): [U, T] {
  return [b, a];
}

// 测试
const [a, b] = swap(1, "hello");
console.log("swap(1, 'hello'):", a, b);  // "hello", 1

const [c, d] = swap(true, 42);
console.log("swap(true, 42):", c, d);    // 42, true

const [e, f] = swap({ name: "Alice" }, [1, 2, 3]);
console.log("swap({...}, [...]):", e, f);

// 进阶：支持数组交换
function swapArrayElements(arr: T[], i: number, j: number): T[] {
  if (i < 0 || i >= arr.length || j < 0 || j >= arr.length) {
    throw new Error("Invalid index");
  }

  const newArr = [...arr];
  [newArr[i], newArr[j]] = [newArr[j], newArr[i]];
  return newArr;
}

const arr = [1, 2, 3, 4, 5];
console.log("swapArrayElements([1,2,3,4,5], 0, 4):", swapArrayElements(arr, 0, 4));

运行结果：

swap(1, 'hello'): hello 1
swap(true, 42): 42 true
swap({...}, [...]): [ 1, 2, 3 ] { name: 'Alice' }
swapArrayElements([1,2,3,4,5], 0, 4): [ 5, 2, 3, 4, 1 ]

练习 7.3：泛型队列类

题目：创建一个泛型类 `Queue`，包含 `enqueue`、`dequeue`、`peek`、`isEmpty` 方法。

TypeScript 代码：

// 泛型队列类
class Queue {
  private items: T[] = [];

  // 入队
  enqueue(item: T): void {
    this.items.push(item);
  }

  // 出队
  dequeue(): T | undefined {
    return this.items.shift();
  }

  // 查看队首元素
  peek(): T | undefined {
    return this.items[0];
  }

  // 队列是否为空
  isEmpty(): boolean {
    return this.items.length === 0;
  }

  // 队列大小
  getSize(): number {
    return this.items.length;
  }

  // 清空队列
  clear(): void {
    this.items = [];
  }

  // 获取所有元素
  toArray(): T[] {
    return [...this.items];
  }

  // 批量入队
  enqueueAll(items: T[]): void {
    this.items.push(...items);
  }
}

// 优先队列（进阶）
class PriorityQueue {
  private items: { item: T; priority: number }[] = [];

  enqueue(item: T, priority: number): void {
    this.items.push({ item, priority });
    this.items.sort((a, b) => a.priority - b.priority);
  }

  dequeue(): T | undefined {
    return this.items.shift()?.item;
  }

  peek(): T | undefined {
    return this.items[0]?.item;
  }

  isEmpty(): boolean {
    return this.items.length === 0;
  }
}

priorityQueue.enqueue("低优先级任务", 3);
priorityQueue.enqueue("高优先级任务", 1);
priorityQueue.enqueue("中优先级任务", 2);

console.log("\n=== 优先队列测试 ===");

  console.log("处理:", priorityQueue.dequeue());
}

练习 7.4：分组函数

题目：创建一个泛型函数 `groupBy`，接受数组和分组函数，返回按分组函数结果组织的对象。

TypeScript 代码：

// 分组函数
function groupBy(
  array: T[],
  keyGetter: (item: T) => K
): Record {
  return array.reduce((groups, item) => {
    const key = keyGetter(item);
    if (!groups[key]) {
      groups[key] = [];
    }
    groups[key].push(item);
    return groups;
  }, {} as Record);
}

// 测试
interface User {
  name: string;
  department: string;
  age: number;
}

const users: User[] = [
  { name: "Alice", department: "Engineering", age: 25 },
  { name: "Bob", department: "Marketing", age: 30 },
  { name: "Charlie", department: "Engineering", age: 28 },
  { name: "Diana", department: "HR", age: 35 },
  { name: "Eve", department: "Marketing", age: 32 }
];

// 按部门分组
const byDepartment = groupBy(users, (user) => user.department);
console.log("按部门分组:");
console.log(byDepartment);

// 按年龄段分组
const byAgeGroup = groupBy(users, (user) => {
  if (user.age < 30) return "young";
  if (user.age < 35) return "middle";
  return "senior";
});

console.log("\n按年龄段分组:");
console.log(byAgeGroup);

// 进阶：支持自定义分隔符
function groupByWithSeparator(
  array: T[],
  keyGetter: (item: T) => string,
  separator: string = "_"
): Record {
  return array.reduce((groups, item) => {
    const key = keyGetter(item).replace(/\s+/g, separator).toLowerCase();
    if (!groups[key]) {
      groups[key] = [];
    }
    groups[key].push(item);
    return groups;
  }, {} as Record);
}

const byFirstLetter = groupByWithSeparator(
  ["Apple", "Banana", "Apricot", "Cherry", "Avocado"],
  (s) => s
);
console.log("\n按首字母分组:");
console.log(byFirstLetter);

练习 7.6：Observable 观察者类

题目：实现一个泛型 `Observable` 类，支持订阅、取消订阅和发布事件。

TypeScript 代码：

// 观察者模式实现
type Subscription = {
  unsubscribe: () => void;
};

type Observer = (data: T) => void;

class Observable {
  private observers: Map[]> = new Map();
  private globalObservers: Observer[] = [];

  // 订阅特定事件
  subscribe(event: string, observer: Observer): Subscription {
    if (!this.observers.has(event)) {
      this.observers.set(event, []);
    }
    this.observers.get(event)!.push(observer);

    return {
      unsubscribe: () => {
        const observers = this.observers.get(event);
        if (observers) {
          const index = observers.indexOf(observer);
          if (index !== -1) {
            observers.splice(index, 1);
          }
        }
      }
    };
  }

  // 订阅所有事件
  subscribeAll(observer: Observer): Subscription {
    this.globalObservers.push(observer);

    return {
      unsubscribe: () => {
        const index = this.globalObservers.indexOf(observer);
        if (index !== -1) {
          this.globalObservers.splice(index, 1);
        }
      }
    };
  }

  // 发布事件
  next(event: string, data: T): void {
    // 通知特定事件订阅者
    const eventObservers = this.observers.get(event);
    if (eventObservers) {
      eventObservers.forEach(observer => observer(data));
    }

    // 通知全局订阅者
    this.globalObservers.forEach(observer => observer(data));
  }

  // 获取观察者数量
  getObserverCount(event?: string): number {
    if (event) {
      return this.observers.get(event)?.length || 0;
    }
    return this.globalObservers.length;
  }
}

// 测试
console.log("=== Observable 测试 ===");

  type: string;
  message: string;
  timestamp: number;
}

const observable = new Observable();

  console.log(`登录事件: ${data.message}`);
});

const sub2 = observable.subscribe("message", (data) => {

});

练习 7.7：泛型表单验证器

题目：创建一个泛型表单验证器类 `FormValidator`。

TypeScript 代码：

// 表单验证器
type ValidationRule = {
  field: keyof T;
  validate: (value: any) => boolean;
  message: string;
};

type ValidationResult = {
  isValid: boolean;
  errors: Partial>;
  data?: T;
};

class FormValidator {
  private rules: ValidationRule[] = [];

  addRule(rule: ValidationRule): this {
    this.rules.push(rule);
    return this;
  }

  addRules(rules: ValidationRule[]): this {
    this.rules.push(...rules);
    return this;
  }

  validate(data: T): ValidationResult {
    const errors: Partial> = {};

    this.rules.forEach(rule => {
      const value = data[rule.field];
      if (!rule.validate(value)) {
        if (!errors[rule.field]) {
          errors[rule.field] = [];
        }
        errors[rule.field]!.push(rule.message);
      }
    });

    return {
      isValid: Object.keys(errors).length === 0,
      errors,
      data: Object.keys(errors).length === 0 ? data : undefined
    };
  }

  // 静态工厂方法
  static create(): FormValidator {
    return new FormValidator();
  }
}

// 预定义验证规则
const Validators = {
  required(field: keyof T, message: string = "此字段必填") {
    return {
      field,
      validate: (value: any) => value !== undefined && value !== null && value !== "",
      message
    };
  },

  minLength(field: keyof T, min: number, message?: string) {
    return {
      field,
      validate: (value: string) => value.length >= min,
      message: message || `最少 ${min} 个字符`
    };
  },

  maxLength(field: keyof T, max: number, message?: string) {
    return {
      field,
      validate: (value: string) => value.length <= max,
      message: message || `最多 ${max} 个字符`
    };
  },

  email(field: keyof T, message: string = "请输入有效的邮箱地址") {
    return {
      field,
      validate: (value: string) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value),
      message
    };
  },

  min(field: keyof T, min: number, message?: string) {
    return {
      field,
      validate: (value: number) => value >= min,
      message: message || `最小值为 ${min}`
    };
  },

  max(field: keyof T, max: number, message?: string) {
    return {
      field,
      validate: (value: number) => value <= max,
      message: message || `最大值为 ${max}`
    };
  },

  pattern(field: keyof T, regex: RegExp, message: string = "格式不正确") {
    return {
      field,
      validate: (value: string) => regex.test(value),
      message
    };
  }
};

// 测试
interface UserForm {
  username: string;
  email: string;
  age: number;
  password: string;
}

console.log("=== FormValidator 测试 ===");

  .addRule(Validators.maxLength("username", 20, "用户名最多20个字符"))
