练习 3.3：学生信息对象

题目：创建一个对象表示学生信息，包含姓名、年龄、课程列表（数组）。使用可选属性表示学生的邮箱。

TypeScript 代码：

// 学生接口定义
interface Student {
  name: string;
  age: number;
  courses: string[];
  email?: string;  // 可选属性
  grade?: number;  // 可选属性
}

// 创建学生对象
const student1: Student = {
  name: "Alice",
  age: 20,
  courses: ["数学", "物理", "化学"]
};

const student2: Student = {
  name: "Bob",
  age: 21,
  courses: ["英语", "历史", "地理"],
  email: "bob@example.com",
  grade: 85
};

// 打印学生信息
function printStudentInfo(student: Student): void {
  console.log(`姓名: ${student.name}`);
  console.log(`年龄: ${student.age}`);
  console.log(`课程: ${student.courses.join(", ")}`);
  if (student.email) {
    console.log(`邮箱: ${student.email}`);
  }
  if (student.grade !== undefined) {
    console.log(`成绩: ${student.grade}分`);
  }
  console.log("---");
}

printStudentInfo(student1);
printStudentInfo(student2);

练习 3.4：Result 联合类型

题目：创建一个类型别名 `Result`，表示可能的值：`"success"`、`"error"` 或 `{code: number, message: string}`。编写一个函数处理这个类型。

TypeScript 代码：

// Result 联合类型定义
type Result =
  | "success"
  | "error"
  | { code: number; message: string };

// 处理 Result 类型的函数
function handleResult(result: Result): string {
  if (result === "success") {
    return "✅ 操作成功！";
  }

  if (result === "error") {
    return "❌ 操作失败！";
  }

  // result 是 { code: number; message: string }
  return `⚠️ [${result.code}] ${result.message}`;
}

// 测试各种情况
const results: Result[] = [
  "success",
  "error",
  { code: 400, message: "请求参数错误" },
  { code: 401, message: "未授权访问" },
  { code: 500, message: "服务器内部错误" }
];

console.log("处理结果：");
results.forEach((result, index) => {
  console.log(`结果${index + 1}: ${handleResult(result)}`);
});

运行结果：

处理结果：
结果1: ✅ 操作成功！
结果2: ❌ 操作失败！
结果3: ⚠️ [400] 请求参数错误
结果4: ⚠️ [401] 未授权访问
结果5: ⚠️ [500] 服务器内部错误

练习 3.5：状态枚举

题目：定义一个状态枚举，包含 `Pending`、`Processing`、`Completed`、`Failed` 四个状态。编写一个函数接受状态并返回对应的描述文字。

TypeScript 代码：

// 状态枚举定义
enum TaskStatus {
  Pending = "PENDING",
  Processing = "PROCESSING",
  Completed = "COMPLETED",
  Failed = "FAILED"
}

// 获取状态描述的函数
function getStatusDescription(status: TaskStatus): string {
  const descriptions: Record = {
    [TaskStatus.Pending]: "⏳ 等待中 - 任务正在等待处理",
    [TaskStatus.Processing]: "🔄 处理中 - 任务正在执行",
    [TaskStatus.Completed]: "✅ 完成 - 任务已成功完成",
    [TaskStatus.Failed]: "❌ 失败 - 任务执行失败"
  };
  return descriptions[status];
}

// 测试
const tasks = [
  { name: "下载文件", status: TaskStatus.Pending },
  { name: "处理数据", status: TaskStatus.Processing },
  { name: "生成报告", status: TaskStatus.Completed },
  { name: "发送邮件", status: TaskStatus.Failed }

tasks.forEach(task => {
  console.log(`${task.name}: ${getStatusDescription(task.status)}`);

练习 3.6：分数转换类型

题目：创建一个表示分数的类型，它可以是以下类型之一：数字、`"A"`、`"B"`、`"C"`、`"D"`、`"F"`。编写一个函数将分数转换为对应的数值（90-100→A，80-89→B，以此类推）。

TypeScript 代码：

// 分数类型定义
type Grade = number | "A" | "B" | "C" | "D" | "F";

// 等级到数值的映射
const gradeToNumber: Record = {
  "A": 90,
  "B": 80,
  "C": 70,
  "D": 60,
  "F": 0
};

// 数值到等级的转换函数
function numericToGrade(score: number): "A" | "B" | "C" | "D" | "F" {
  if (score >= 90) return "A";
  if (score >= 80) return "B";
  if (score >= 70) return "C";
  if (score >= 60) return "D";
  return "F";
}

// 等级到数值的转换函数
function gradeToNumeric(grade: string): number {
  return gradeToNumber[grade] ?? 0;
}

// 综合处理函数
function processGrade(grade: Grade): { grade: string; numeric: number; description: string } {
  let numeric: number;
  let description: string;

  if (typeof grade === "number") {
    numeric = grade;
  } else {
    numeric = gradeToNumeric(grade);
  }

  const letterGrade = numericToGrade(numeric);

  switch (letterGrade) {
    case "A": description = "优秀"; break;
    case "B": description = "良好"; break;
    case "C": description = "中等"; break;
    case "D": description = "及格"; break;
    case "F": description = "不及格"; break;
  }

  return {
    grade: letterGrade,
    numeric: numeric,
    description: description
  };
}

// 测试
const scores: Grade[] = [95, 88, 78, 65, 42, "A", "B", "C", "D", "F"];

console.log("分数转换系统：");
console.log("================");
scores.forEach(score => {
  const result = processGrade(score);
  console.log(`输入: ${typeof score === "number" ? score : `"${score}"`} → 等级: ${result.grade}, 分数: ${result.numeric}, 评价: ${result.description}`);
});

运行结果：

输入: 88 → 等级: B, 分数: 88, 评价: 良好
输 入: 78 → 等级: C, 分数: 78, 评价: 中等
输入: 65 → 等级: D, 分数: 65, 评价: 及格

练习 3.7：可选属性工具类型

题目：创建一个工具类型，将对象的所有属性变为可选，并创建一个函数演示其用法。

TypeScript 代码：

// 工具类型：将 T 的所有属性变为可选
type Partial = {
  [P in keyof T]?: T[P];
};

// 用户类型
interface User {
  id: string;
  name: string;
  email: string;
  age: number;
  address: string;
  phone: string;
}

// 使用 Partial 创建可选类型
type PartialUser = Partial;

// 演示函数
function createUser(defaults: PartialUser): User {
  // 使用默认值填充缺失的属性
  return {
    id: defaults.id || generateId(),
    name: defaults.name || "Unknown User",
    email: defaults.email || "",
    age: defaults.age || 0,
    address: defaults.address || "",
    phone: defaults.phone || ""
  };
}

// 辅助函数：生成随机 ID
function generateId(): string {
  return "user_" + Math.random().toString(36).substr(2, 9);
}

// 测试
console.log("使用 Partial 类型创建用户：");

// 创建完整用户
const fullUser = createUser({
  id: "user-001",
  name: "Alice",
  email: "alice@example.com",
  age: 25
});
console.log("完整用户:", fullUser);

// 创建部分用户
const partialUser = createUser({
  name: "Bob"
});
console.log("部分用户:", partialUser);

// 创建空用户
const emptyUser = createUser({});
console.log("空用户:", emptyUser);

// 验证原始类型仍然需要所有属性
// const test: User = { name: "Test" };  // 错误：缺少属性
