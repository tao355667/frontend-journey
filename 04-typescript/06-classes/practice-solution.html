练习 6.1：银行账户类

题目：创建一个 `BankAccount` 类，包含私有属性 `balance`，公共方法 `deposit`、`withdraw` 和 `getBalance`。

TypeScript 代码：

// 银行账户类
class BankAccount {
  private balance: number;
  private readonly accountNumber: string;
  private readonly accountHolder: string;
  private transactions: { type: 'deposit' | 'withdraw'; amount: number; date: Date }[] = [];

  constructor(accountHolder: string, accountNumber: string, initialBalance: number = 0) {
    this.accountHolder = accountHolder;
    this.accountNumber = accountNumber;
    this.balance = initialBalance;
    this.transactions.push({ type: 'deposit', amount: initialBalance, date: new Date() });
  }

  // 存款方法
  public deposit(amount: number): boolean {
    if (amount <= 0) {
      console.log("存款金额必须大于0");
      return false;
    }
    this.balance += amount;
    this.transactions.push({ type: 'deposit', amount, date: new Date() });
    console.log(`存款 ${amount} 成功，当前余额：${this.balance}`);
    return true;
  }

  // 取款方法
  public withdraw(amount: number): boolean {
    if (amount <= 0) {
      console.log("取款金额必须大于0");
      return false;
    }
    if (amount > this.balance) {
      console.log(`取款失败，余额不足。当前余额：${this.balance}`);
      return false;
    }
    this.balance -= amount;
    this.transactions.push({ type: 'withdraw', amount, date: new Date() });
    console.log(`取款 ${amount} 成功，当前余额：${this.balance}`);
    return true;
  }

  // 获取余额
  public getBalance(): number {
    return this.balance;
  }

  // 获取账户信息
  public getAccountInfo(): string {
    return `账户：${this.accountNumber}，持有人：${this.accountHolder}，余额：${this.balance}`;
  }

  // 获取交易记录
  public getTransactions(): { type: 'deposit' | 'withdraw'; amount: number; date: Date }[] {
    return [...this.transactions];
  }
}

练习 6.2：矩形类

题目：创建一个 `Rectangle` 类，包含属性 `width` 和 `height`，以及方法 `getArea` 和 `getPerimeter`。

TypeScript 代码：

// 矩形类
class Rectangle {
  constructor(public width: number, public height: number) {}

  public getArea(): number {
    return this.width * this.height;
  }

  public getPerimeter(): number {
    return 2 * (this.width + this.height);
  }

  public getDiagonal(): number {
    return Math.sqrt(this.width ** 2 + this.height ** 2);
  }

  public isSquare(): boolean {
    return this.width === this.height;
  }

  public scale(factor: number): void {
    if (factor <= 0) {
      console.log("缩放因子必须大于0");
      return;
    }
    this.width *= factor;
    this.height *= factor;
  }

  public toString(): string {
    if (this.isSquare()) {
      return `正方形 (${this.width}x${this.height})`;
    }
    return `矩形 (${this.width}x${this.height})`;
  }
}

练习 6.3：人物自我介绍类

题目：创建一个 `Person` 类，包含属性 `name` 和 `age`，以及方法 `introduce` 返回自我介绍字符串。

TypeScript 代码：

// 人物类
class Person {
  private static readonly ADULT_AGE = 18;

  constructor(
    public name: string,
    public age: number,
    private occupation?: string
  ) {}

  public introduce(): string {
    let intro = `我叫${this.name}，今年${this.age}岁`;
    
    if (this.occupation) {
      intro += `，是一名${this.occupation}`;
    }
    
    if (this.age >= Person.ADULT_AGE) {
      intro += "，已成年";
    } else {
      intro += "，未成年";
    }
    
    return intro;
  }

  public hasBirthday(): void {
    this.age++;
    console.log(`生日快乐！${this.name} 现在 ${this.age} 岁了`);
  }

  public getOccupation(): string | undefined {
    return this.occupation;
  }

  public setOccupation(occupation: string): void {
    this.occupation = occupation;
  }

  public static isAdult(age: number): boolean {
    return age >= Person.ADULT_AGE;
  }
}

练习 6.5：抽象图形类

题目：创建一个抽象类 `Shape`，包含抽象方法 `getArea`。创建 `Circle` 和 `Rectangle` 子类实现这个方法。

TypeScript 代码：

// 抽象图形类
abstract class Shape {
  constructor(public color: string) {}

  // 抽象方法：必须在子类中实现
  abstract getArea(): number;
  abstract getPerimeter(): number;

  // 具体方法
  public describe(): string {
    return `一个${this.color}的${this.constructor.name}，面积=${this.getArea().toFixed(2)}`;
  }

  public getInfo(): string {
    return `${this.describe()}，周长=${this.getPerimeter().toFixed(2)}`;
  }
}

// 圆形类
class Circle extends Shape {
  constructor(
    public radius: number,
    color: string
  ) {
    super(color);
  }

  getArea(): number {
    return Math.PI * this.radius * this.radius;
  }

  getPerimeter(): number {
    return 2 * Math.PI * this.radius;
  }

  getDiameter(): number {
    return this.radius * 2;
  }
}

// 矩形类
class Rectangle extends Shape {
  constructor(
    public width: number,
    public height: number,
    color: string
  ) {
    super(color);
  }

  getArea(): number {
    return this.width * this.height;
  }

  getPerimeter(): number {
    return 2 * (this.width + this.height);
  }

  isSquare(): boolean {
    return this.width === this.height;
  }
}

// 三角形类
class Triangle extends Shape {
  constructor(
    public sideA: number,
    public sideB: number,
    public sideC: number,
    color: string
  ) {
    super(color);
  }

  getArea(): number {
    // 海伦公式
    const s = (this.sideA + this.sideB + this.sideC) / 2;
    return Math.sqrt(s * (s - this.sideA) * (s - this.sideB) * (s - this.sideC));
  }

  getPerimeter(): number {
    return this.sideA + this.sideB + this.sideC;
  }

  isEquilateral(): boolean {
    return this.sideA === this.sideB && this.sideB === this.sideC;
  }

  isRightTriangle(): boolean {
    const sides = [this.sideA, this.sideB, this.sideC].sort((a, b) => a - b);
    return Math.abs(sides[0] ** 2 + sides[1] ** 2 - sides[2] ** 2) < 0.0001;
  }
}

练习 6.6：图书管理系统

题目：创建一个图书管理系统，包含 `Book`、`Member` 和 `Library` 类。

TypeScript 代码：

// 图书类
class Book {
  constructor(
    public readonly isbn: string,
    public title: string,
    public author: string,
    public year: number,
    public available: boolean = true
  ) {}

  public borrow(): boolean {
    if (!this.available) {
      console.log(`《${this.title}》已被借出`);
      return false;
    }
    this.available = false;
    return true;
  }

  public return(): void {
    this.available = true;
    console.log(`《${this.title}》已归还`);
  }

  public getInfo(): string {
    const status = this.available ? "可借阅" : "已借出";
    return `《${this.title}》- ${this.author} (${this.year}) [${status}]`;
  }
}

// 会员类
class Member {
  private borrowedBooks: Book[] = [];
  private readonly maxBooks = 5;

  constructor(
    public readonly memberId: string,
    public name: string,
    public email: string
  ) {}

  public borrowBook(book: Book): boolean {
    if (this.borrowedBooks.length >= this.maxBooks) {
      console.log(`${this.name} 已达到最大借阅数量（${this.maxBooks}本）`);
      return false;
    }

    if (!book.borrow()) {
      return false;
    }

    this.borrowedBooks.push(book);
    console.log(`${this.name} 借阅了《${book.title}》`);
    return true;
  }

  public returnBook(book: Book): boolean {
    const index = this.borrowedBooks.findIndex(b => b.isbn === book.isbn);
    if (index === -1) {
      console.log(`${this.name} 没有借阅《${book.title}》`);
      return false;
    }

    this.borrowedBooks.splice(index, 1);
    book.return();
    return true;
  }

  public getBorrowedBooks(): Book[] {
    return [...this.borrowedBooks];
  }

  public getBorrowedCount(): number {
    return this.borrowedBooks.length;
  }

  public getInfo(): string {
    return `${this.name} (ID: ${this.memberId}) - 已借 ${this.borrowedBooks.length}/${this.maxBooks} 本`;
  }
}

// 图书馆类
class Library {
  private books: Map = new Map();
  private members: Map = new Map();

  public addBook(book: Book): void {
    this.books.set(book.isbn, book);
    console.log(`图书馆新增：《${book.title}》`);
  }

  public addMember(member: Member): void {
    this.members.set(member.memberId, member);
    console.log(`图书馆新增会员：${member.name}`);
  }

  public findBook(isbn: string): Book | undefined {
    return this.books.get(isbn);
  }

  public findMember(memberId: string): Member | undefined {
    return this.members.get(memberId);
  }

  public lendBook(memberId: string, isbn: string): boolean {
    const member = this.members.get(memberId);
    const book = this.books.get(isbn);

    if (!member) {
      console.log("会员不存在");
      return false;
    }

    if (!book) {
      console.log("图书不存在");
      return false;
    }

    return member.borrowBook(book);
  }

  public returnBook(memberId: string, isbn: string): boolean {
    const member = this.members.get(memberId);
    const book = this.books.get(isbn);

    if (!member || !book) {
      console.log("会员或图书不存在");
      return false;
    }

    return member.returnBook(book);
  }

  public getAvailableBooks(): Book[] {
    return Array.from(this.books.values()).filter(b => b.available);
  }

  public getAllBooks(): Book[] {
    return Array.from(this.books.values());
  }

  public getAllMembers(): Member[] {
    return Array.from(this.members.values());
  }

  public getStatistics(): {
    totalBooks: number;
    availableBooks: number;
    totalMembers: number;
    borrowedBooks: number;
  } {
    const allBooks = this.getAllBooks();
    return {
      totalBooks: allBooks.length,
      availableBooks: allBooks.filter(b => b.available).length,
      totalMembers: this.members.size,
      borrowedBooks: allBooks.filter(b => !b.available).length
    };
  }
}
