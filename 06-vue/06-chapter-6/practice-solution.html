// 主题 Class 绑定 <div class="theme-container" :class="themeClass"> // 计算属性 data() { return { currentTheme: 'light', followSystemTheme: false } }, computed: { themeClass() { return `theme-${this.currentTheme}` } }, methods: { setTheme(theme) { this.currentTheme = theme this.followSystemTheme = false localStorage.setItem('theme', theme) }, followSystem() { this.followSystemTheme = true this.currentTheme = this.getSystemTheme() // 监听系统主题变化 window.matchMedia('(prefers-color-scheme: dark)') .addEventListener('change', e => { if (this.followSystemTheme) { this.currentTheme = e.matches ? 'dark' : 'light' } }) } } // CSS 变量使用 .theme-light { --bg-color: #ffffff; --text-color: #333333; } .theme-dark { --bg-color: #1a1a2e; --text-color: #eeeeee; }
