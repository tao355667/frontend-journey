{{ item.name }} × {{ item.quantity }} = ¥{{ item.price * item.quantity }}

应付总额: ¥{{ finalPrice }}

// TypeScript 计算属性 - 显式返回类型 computed: { // 返回 number 类型 totalItems(): number { return this.cart.reduce((sum: number, item: CartItem) => { return sum + item.quantity }, 0) }, // 返回 number 类型 subtotal(): number { return this.cart.reduce((sum: number, item: CartItem) => { return sum + item.price * item.quantity }, 0) }, // 返回 string 类型（格式化后的价格） finalPrice(): string { const price: number = this.isVip ? this.subtotal * 0.9 : this.subtotal return price.toFixed(2) } }

可写计算属性的类型定义

// 可写计算属性的完整类型 data() { return { rmbPrice: 720 // number 类型 } }, computed: { usdPrice: { // getter 返回 number get(): number { return Number((this.rmbPrice / 7.2).toFixed(2)) }, // setter 接收 number 类型参数 set(value: number): void { this.rmbPrice = Number((value * 7.2).toFixed(2)) } } }
