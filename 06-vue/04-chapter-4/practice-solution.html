核心计算属性：

computed: { totalScore() { return this.scores.reduce((sum, score) => sum + score, 0) }, averageScore() { return (this.totalScore / this.scores.length).toFixed(1) }, maxScore() { return Math.max(...this.scores) }, minScore() { return Math.min(...this.scores) }, allPassed() { return this.scores.every(score => score >= 60) }, overallGrade() { const avg = parseFloat(this.averageScore) if (avg >= 90) return 'A' if (avg >= 80) return 'B' if (avg >= 70) return 'C' if (avg >= 60) return 'D' return 'F' } }

computed: { filteredProducts() { let result = this.products // 价格筛选 if (this.minPrice > 0) { result = result.filter(p => p.price >= this.minPrice) } if (this.maxPrice > 0) { result = result.filter(p => p.price <= this.maxPrice) } // 类别筛选 if (this.selectedCategory) { result = result.filter(p => p.category === this.selectedCategory) } // 库存筛选 if (this.showInStockOnly) { result = result.filter(p => p.stock > 0) } // 排序 switch(this.sortBy) { case 'price-asc': result = result.sort((a, b) => a.price - b.price) break case 'price-desc': result = result.sort((a, b) => b.price - a.price) break case 'stock': result = result.sort((a, b) => b.stock - a.stock) break } return result }, totalFilteredStock() { return this.filteredProducts.reduce((sum, p) => sum + p.stock, 0) } }
