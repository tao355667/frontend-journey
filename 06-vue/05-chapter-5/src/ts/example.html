TypeScript 类型定义

// TypeScript 接口定义 interface User { name: string age: number email: string } interface ValidationState { valid: boolean message: string } interface AppData { user: User validation: { name: ValidationState email: ValidationState } searchQuery: string searchResults: SearchResult[] } interface SearchResult { id: number title: string } // Watch 处理器类型 interface WatchHandler { (newVal: T, oldVal: T): void }

带类型的侦听器

当前用户: {{ user.name }} ({{ user.age }}岁)

// TypeScript 带类型参数的侦听器 watch: { // 基础类型侦听 'user.name'(newName: string, oldName: string) { console.log(`姓名变化: ${oldName} → ${newName}`) }, // 对象类型侦听 user: { handler(newUser: User, oldUser: User) { console.log('用户对象变化:', newUser) }, deep: true }, // 数组类型侦听 searchResults(newResults: SearchResult[], oldResults: SearchResult[]) { console.log(`结果数量: ${oldResults.length} → ${newResults.length}`) } }
