data() { return { current: '0', // 当前输入 previous: null, // 前一个数字 operator: null, // 运算符 shouldReset: false // 是否重置显示 } }, methods: { inputNumber(num) { if (this.shouldReset) { this.current = num this.shouldReset = false } else { this.current = this.current === '0' ? num : this.current + num } }, calculate() { const prev = parseFloat(this.previous) const current = parseFloat(this.current) switch(this.operator) { case '+': this.current = (prev + current).toString(); break case '-': this.current = (prev - current).toString(); break case '*': this.current = (prev * current).toString(); break case '/': this.current = current !== 0 ? (prev / current).toString() : 'Error' } } }

data() { return { todos: [], newTodoText: '', nextId: 1 } }, computed: { completedCount() { return this.todos.filter(t => t.completed).length } }, methods: { addTodo() { if (!this.newTodoText.trim()) return this.todos.push({ id: this.nextId++, text: this.newTodoText, completed: false }) this.newTodoText = '' }, removeTodo(id) { this.todos = this.todos.filter(t => t.id !== id) }, toggleTodo(id) { const todo = this.todos.find(t => t.id === id) if (todo) todo.completed = !todo.completed }, clearCompleted() { this.todos = this.todos.filter(t => !t.completed) } }

data() { return { products: [ { id: 1, name: '笔记本电脑', price: 5999, stock: 5 }, { id: 2, name: '无线鼠标', price: 99, stock: 20 }, { id: 3, name: '机械键盘', price: 399, stock: 10 }, { id: 4, name: '显示器', price: 1299, stock: 8 } ], cart: { items: [] // { productId, name, price, quantity } } } }, methods: { addToCart(product) { const existingItem = this.cart.items.find( item => item.productId === product.id ) if (existingItem) { existingItem.quantity++ } else { this.cart.items.push({ productId: product.id, name: product.name, price: product.price, quantity: 1 }) } product.stock-- }, removeFromCart(productId) { const item = this.cart.items.find(i => i.productId === productId) const product = this.products.find(p => p.id === productId) if (item && product) { product.stock += item.quantity } this.cart.items = this.cart.items.filter( i => i.productId !== productId ) }, updateQuantity(productId, quantity) { if (quantity <= 0) { this.removeFromCart(productId) return } const item = this.cart.items.find(i => i.productId === productId) const product = this.products.find(p => p.id === productId) if (item && product) { const diff = quantity - item.quantity if (diff > 0 && product.stock >= diff) { item.quantity = quantity product.stock -= diff } else if (diff < 0) { item.quantity = quantity product.stock -= diff // diff 是负数，所以是加回来 } } }, getCartTotal() { return this.cart.items.reduce((total, item) => { return total + item.price * item.quantity }, 0).toFixed(2) }, getCartItemCount() { return this.cart.items.reduce((count, item) => count + item.quantity, 0) }, checkout() { if (this.cart.items.length === 0) { alert('购物车是空的！') return } const total = this.getCartTotal() alert(`订单提交成功！\n商品数量: ${this.getCartItemCount()}\n总金额: ¥${total}`) this.cart.items = [] } }
