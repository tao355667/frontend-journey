// TypeScript 方法定义 methods: { // 无参数，无返回值 incrementCount(): void { this.count++ }, // 带参数，无返回值 setUserName(name: string): void { this.user.name = name }, // 带返回值 getProductCount(): number { return this.products.length }, // 复杂参数 addProduct(product: Product): void { this.products.push(product) } }

3. 完整购物车示例

购物车

会员状态: {{ cart.isVip ? 'VIP会员' : '普通用户' }}

商品数量: {{ cart.items.length }}

总价: ¥{{ cart.total }}

{{ item.name }} ¥{{ item.price }} × {{ item.quantity }}
// TypeScript 购物车方法 methods: { addToCart(product: Product): void { const existingItem = this.cart.items.find( (item: CartItem) => item.name === product.name ) if (existingItem) { existingItem.quantity++ } else { this.cart.items.push({ ...product, quantity: 1 } as CartItem) } this.updateCartTotal() }, updateCartTotal(): void { let total: number = this.cart.items.reduce( (sum: number, item: CartItem) => sum + item.price * item.quantity, 0 ) if (this.cart.isVip) { total = total * 0.9 } this.cart.total = Number(total.toFixed(2)) } }
